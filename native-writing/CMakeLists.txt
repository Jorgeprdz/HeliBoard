cmake_minimum_required(VERSION 3.22)
project(heliboard_writing LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" llama)
if(ANDROID)
 add_library(heliboard_writing SHARED writing.cpp)
 target_compile_options(heliboard_writing PRIVATE -fvisibility=hidden)
 target_link_options(heliboard_writing PRIVATE -Wl,-z,max-page-size=16384)
else()
 add_executable(writing-eval writing.cpp)
endif()
if(ANDROID)
 target_link_libraries(heliboard_writing PRIVATE llama)
else()
 target_link_libraries(writing-eval PRIVATE llama)
endif()
